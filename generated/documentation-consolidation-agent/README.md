export function fillTemplate(template: string, category: DocumentCategory): string {
  let result &#x3D; template;
  
  // Extract all placeholders
  const placeholders &#x3D; template.match(/\{\{([^}]+)\}\}/g) || [];
  
  placeholders.forEach(placeholder &#x3D;&gt; {
    const key &#x3D; placeholder.replace(/[{}]/g, &#x27;&#x27;);
    let content &#x3D; &#x27;&#x27;;
    
    // Skip references placeholder - will be filled later
    if (key &#x3D;&#x3D;&#x3D; &#x27;references&#x27;) return;
    
    // For each placeholder, find relevant content from documents
    category.documents.forEach(doc &#x3D;&gt; {
      doc.sections.forEach(section &#x3D;&gt; {
        const processSection &#x3D; (sect: any, depth &#x3D; 0) &#x3D;&gt; {
          // Check if section title matches placeholder
          if (sect.title.toLowerCase().includes(key.toLowerCase())) {
            content +&#x3D; &#x60;${sect.content}\n\n&#x60;;
          }
          
          // Process subsections
          if (sect.subsections &amp;&amp; sect.subsections.length &gt; 0) {
            sect.subsections.forEach((subsect: any) &#x3D;&gt; processSection(subsect, depth + 1));
          }
        };
        
        processSection(section);
      });
    });
    
    // If no specific content found, use general content from top documents
    if (!content &amp;&amp; key !&#x3D;&#x3D; &#x27;title&#x27; &amp;&amp; key !&#x3D;&#x3D; &#x27;overview&#x27;) {
      const topDocs &#x3D; category.documents.slice(0, 3); // Use top 3 docs
      topDocs.forEach(doc &#x3D;&gt; {
        if (doc.sections.length &gt; 0) {
          content +&#x3D; &#x60;### From ${doc.metadata.filename}\n\n&#x60;;
          content +&#x3D; doc.sections[0].content + &#x27;\n\n&#x27;;
        }
      });
    }
    
    // For overview, use document themes and first paragraphs
    if (key &#x3D;&#x3D;&#x3D; &#x27;overview&#x27; &amp;&amp; !content) {
      content &#x3D; &#x60;This document consolidates information about ${category.name} from multiple sources.\n\n&#x60;;
      content +&#x3D; &#x60;Key topics include: ${category.documents.flatMap(d &#x3D;&gt; d.themes).slice(0, 5).join(&#x27;, &#x27;)}.\n\n&#x60;;
      
      // Add first paragraph from top document
      if (category.documents.length &gt; 0 &amp;&amp; category.documents[0].sections.length &gt; 0) {
        const firstPara &#x3D; category.documents[0].sections[0].content.split(&#x27;\n\n&#x27;)[0];
        if (firstPara) content +&#x3D; firstPara + &#x27;\n\n&#x27;;
      }
    }
    
    result &#x3D; result.replace(placeholder, content.trim());
  });
  
  return result;
}

export async function generateDocument(category: DocumentCategory, config: GenerationConfig): Promise&lt;GeneratedDocument&gt; {
  // Step 1: Get template for this category
  const template &#x3D; getTemplateForCategory(config.category);
  
  // Step 2: Fill template with content
  let content &#x3D; fillTemplate(template, category);
  
  // Step 3: Generate footnotes for all documents in category
  const allFootnotes: FootnoteReference[] &#x3D; [];
  
  category.documents.forEach(doc &#x3D;&gt; {
    const { footnotes } &#x3D; generateFootnotes(content, doc);
    allFootnotes.push(...footnotes);
  });
  
  // Step 4: Check if target file exists and has footnotes
  let existingFootnotes: FootnoteReference[] &#x3D; [];
  
  if (config.preserveExisting &amp;&amp; await fs.pathExists(config.targetFile)) {
    const existingContent &#x3D; await fs.readFile(config.targetFile, &#x27;utf-8&#x27;);
    existingFootnotes &#x3D; extractExistingFootnotes(existingContent);
    
    // Preserve existing content structure but update with new information
    if (existingContent.length &gt; 0) {
      // Simple merge strategy - keep existing content but add new sections
      const existingSections &#x3D; existingContent.split(&#x27;\n## &#x27;);
      const newSections &#x3D; content.split(&#x27;\n## &#x27;);
      
      // Combine sections, preferring existing ones but adding new ones
      const combinedSections &#x3D; [...existingSections];
      
      newSections.forEach(newSection &#x3D;&gt; {
        const sectionTitle &#x3D; newSection.split(&#x27;\n&#x27;)[0];
        const existingSectionIndex &#x3D; existingSections.findIndex(s &#x3D;&gt; 
          s.split(&#x27;\n&#x27;)[0].toLowerCase() &#x3D;&#x3D;&#x3D; sectionTitle.toLowerCase()
        );
        
        if (existingSectionIndex &#x3D;&#x3D;&#x3D; -1) {
          // This is a new section, add it
          combinedSections.push(newSection);
        }
      });
      
      // Rebuild content, ensuring first section doesn&#x27;t get &#x27;## &#x27; prefix
      content &#x3D; combinedSections[0] + combinedSections.slice(1).map(s &#x3D;&gt; &#x60;\n## ${s}&#x60;).join(&#x27;&#x27;);
    }
  }
  
  // Step 5: Merge footnotes and apply to content
  const mergedFootnotes &#x3D; mergeFootnotes(existingFootnotes, allFootnotes);
  const finalContent &#x3D; applyFootnotesToContent(content, mergedFootnotes);
  
  return {
    path: config.targetFile,
    content: finalContent,
    footnotes: mergedFootnotes
  };
}

import fs from &#x27;fs-extra&#x27;;
import path from &#x27;path&#x27;;

export async function detectOutOfSync(sources: string[], targets: Record&lt;string, string&gt;): Promise&lt;boolean&gt; {
  let isOutOfSync &#x3D; false;
  
  // Check if any source files are newer than target files
  for (const sourceDir of sources) {
    if (!await fs.pathExists(sourceDir)) continue;
    
    const sourceFiles &#x3D; await fs.readdir(sourceDir);
    
    for (const sourceFile of sourceFiles) {
      const sourcePath &#x3D; path.join(sourceDir, sourceFile);
      const sourceStat &#x3D; await fs.stat(sourcePath);
      
      // Skip directories
      if (sourceStat.isDirectory()) continue;
      
      // Check against each target file
      for (const targetFile of Object.values(targets)) {
        if (!await fs.pathExists(targetFile)) {
          isOutOfSync &#x3D; true;
          console.log(&#x60;Target file ${targetFile} does not exist - consolidation needed&#x60;);
          continue;
        }
        
        const targetStat &#x3D; await fs.stat(targetFile);
        
        if (sourceStat.mtime &gt; targetStat.mtime) {
          isOutOfSync &#x3D; true;
          console.log(&#x60;Source file ${sourcePath} is newer than target ${targetFile} - consolidation needed&#x60;);
        }
      }
    }
  }
  
  return isOutOfSync;
}

# Documentation Consolidation Agent

Automatically consolidate scattered documentation files into comprehensive root documentation with proper cross-references.

## Overview

The Documentation Consolidation Agent is designed to solve the problem of fragmented documentation across multiple directories. It scans source directories, analyzes document content, and generates consolidated documentation files with proper cross-references and footnotes.

## Features

- Scans multiple source directories for documentation files
- Categorizes documents by type (system guide, troubleshooting, setup, etc.)
- Extracts key information while preserving context
- Maintains proper numbered footnoting system in [1], [2], [3] format
- Ensures zero information loss from source documents
- Generates comprehensive root documentation files
- Supports multiple documentation formats (.md, .txt)
